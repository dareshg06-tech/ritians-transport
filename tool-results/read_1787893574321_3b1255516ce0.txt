    29→function lerp(a: Coord, b: Coord, t: number): Coord {
    30→  return { lat: a.lat + (b.lat - a.lat) * t, lng: a.lng + (b.lng - a.lng) * t };
    31→}
    32→
    33→function haversineDist(a: Coord, b: Coord): number {
    34→  const R = 6371000;
    35→  const toRad = (d: number) => (d * Math.PI) / 180;
    36→  const dLat = toRad(b.lat - a.lat);
    37→  const dLng = toRad(b.lng - a.lng);
    38→  const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(dLng / 2) ** 2;
    39→  return 2 * R * Math.asin(Math.sqrt(h));
    40→}

    42→// Track ALL routes — user wants "where is my bus" for all buses
    43→const TRACKED_ROUTE_NOS = ALL_ROUTES.map((r) => r.routeNo);
    44→
    45→function initialBuses(): BusPosition[] {
    46→  return TRACKED_ROUTE_NOS.map((rno) => {
    47→    const r = ALL_ROUTES.find((x) => x.routeNo === rno)!;
    48→    const stops = routeStops[rno] || [];
    49→    const firstStop = stops[0];
    50→    const startCoords = firstStop?.coords || r.coords;
    51→    return { routeNo: r.routeNo, routeName: r.routeName, no: r.no, coords: startCoords, progress: 0, segIdx: 0, speed: 35 };
    52→  });
    53→}

   420→                    const distFromPrev = haversine(prevCoords, sc);
   421→                    // Per-stop ETA based on distance from bus + speed
   422→                    const stopEta = distFromBus < 150 ? 0 : Math.max(1, Math.round((distFromBus / 1000) / (Math.max(selectedBus!.speed, 20) / 60)));

   441→                        </div>
   442→                        {/* Stop name + time */}
   443→                        <div style={{ flex: 1, minWidth: 0 }}>
   444→                          <span style={{
   445→                            fontSize: 12, fontWeight: isCurrent ? 700 : 500,
   446→                            color: isCrossed ? "#10b981" : isCurrent ? "#06b6d4" : isFinal ? "#f59e0b" : "#94a3b8",
   447→                          }}>
   448→                            {i + 1}. {s.stop}
   449→                            {isFinal && <span style={{ marginLeft: 6, fontSize: 9, fontWeight: 700, color: "#f59e0b", padding: "1px 5px", borderRadius: 99, background: "rgba(245,158,11,0.1)" }}>FINAL</span>}
   450→                          </span>
   451→                          <span style={{ fontSize: 10, color: "#64748b", marginLeft: 8, fontFamily: "var(--font-mono)" }}>{s.time}</span>
   452→                        </div>
   453→                        {/* Distance from previous stop */}
   454→                        <span style={{ fontSize: 10, color: "#64748b", fontFamily: "var(--font-mono)" }}>
   455→                          {i > 0 ? `${(distFromPrev / 1000).toFixed(1)} km` : "—"}
   456→                        </span>
   457→                        {/* Status badge */}
   458→                        {isCrossed ? (
   459→                          <span style={{ fontSize: 9, fontWeight: 700, color: "#10b981", padding: "2px 6px", borderRadius: 99, background: "rgba(16,185,129,0.1)" }}>CROSSED</span>
   460→                        ) : isCurrent ? (
   461→                          <span style={{ fontSize: 9, fontWeight: 700, color: "#06b6d4", padding: "2px 6px", borderRadius: 99, background: "rgba(6,182,212,0.1)" }}>HERE NOW</span>
   462→                        ) : (
   463→                          <span style={{ fontSize: 9, fontWeight: 700, color: stopEta > 0 ? "#06b6d4" : "#64748b", padding: "2px 6px", borderRadius: 99, background: stopEta > 0 ? "rgba(6,182,212,0.08)" : "rgba(255,255,255,0.03)" }}>
   464→                            {stopEta > 0 ? `${stopEta} min` : "—"}
   465→                          </span>
   466→                        )}
   467→                      </div>
   468→                    );
   469→                  })}
   470→                </div>
   471→              </div>
   472→            );
   473→          })()}
   474→
   475→          {/* Route selector */}
   476→          <div style={{ marginBottom: 14 }}>
   477→            <div style={{ fontSize: 11, fontWeight: 700, textTransform: "uppercase", letterSpacing: "0.05em", color: "#06b6d4", marginBottom: 8 }}>
   478→              <i className="fas fa-route" /> Select Route
   479→            </div>
   480→            <select
   481→              value={selectedRoute}
   482→              onChange={(e) => setSelectedRoute(e.target.value)}
   483→              style={{
   484→                width: "100%", padding: "10px 14px",
   485→                background: "#13161c", border: "1px solid #1e2330", borderRadius: 8,

   541→                      </div>
   542→                    </div>

   546→                        <span style={{ width: 6, height: 6, borderRadius: "50%", background: "#06b6d4", boxShadow: "0 0 6px #06b6d4" }} /> DRIVER GPS

   549→                      <div style={{ display: "flex", alignItems: "center", gap: 5, padding: "3px 8px", borderRadius: 99, background: "rgba(16,185,129,0.15)", border: "1px solid rgba(16,185,129,0.3)", fontSize: 10, fontWeight: 700, color: "#10b981", flexShrink: 0 }}>

   554→                        <span style={{ width: 6, height: 6, borderRadius: "50%", background: "#ef4444" }} /> OFF
