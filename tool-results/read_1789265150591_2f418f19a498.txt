    27→function lerp(a: Coord, b: Coord, t: number): Coord {
    28→  return { lat: a.lat + (b.lat - a.lat) * t, lng: a.lng + (b.lng - a.lng) * t };
    29→}
    30→function haversineDist(a: Coord, b: Coord): number {
    31→  const R = 6371000;
    32→  const toRad = (d: number) => (d * Math.PI) / 180;
    33→  const dLat = toRad(b.lat - a.lat);
    34→  const dLng = toRad(b.lng - a.lng);
    35→  const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(dLng / 2) ** 2;
    36→  return 2 * R * Math.asin(Math.sqrt(h));
    37→}

   138→function DriverModePanel({
   139→  driverPositions,
   140→  tick,
   141→  isMobile = false,
   142→}: {
   143→  driverPositions: Record<string, { coords: Coord; speed: number; heading: number | null; timestamp: number }>;
   144→  tick: number;
   145→  isMobile?: boolean;
   146→}) {
   147→  const { show } = useToast();
   148→  const routeToVehicleId = useVehicleIdMap();
   149→  const [selectedRoute, setSelectedRoute] = useState("");
   150→  const [sharing, setSharing] = useState(false);
   151→  const [gps, setGps] = useState<DriverGPSData | null>(null);
   152→  const [error, setError] = useState<string | null>(null);
   153→  const [positionSource, setPositionSource] = useState<"gps" | "network">("gps");
   154→  const [tripStartTime, setTripStartTime] = useState<number | null>(null);
   155→  const [usingSimulated, setUsingSimulated] = useState(false);
   156→  const watchIdRef = useRef<number | null>(null);
   157→  const simIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   158→  const segIdxRef = useRef(0);
   159→  const progRef = useRef(0);
   160→
   161→  const vehicle = ALL_ROUTES.find((r) => r.routeNo === selectedRoute);
   162→  const dbVehicleId = routeToVehicleId[selectedRoute] || null;
   163→
   164→  const postLocation = (data: DriverGPSData, isSim: boolean) => {
   165→    if (!dbVehicleId) return;
   166→    fetch("/api/locations", {
   167→      method: "POST",
   168→      headers: { "Content-Type": "application/json" },
   169→      body: JSON.stringify({
   170→        vehicleId: dbVehicleId,
   171→        latitude: data.latitude,
   172→        longitude: data.longitude,
   173→        accuracy: data.accuracy,
   174→        speed: data.speed,
   175→        heading: data.heading,
   176→        altitude: data.altitude,
   177→        isSimulated: isSim,
   178→      }),
   179→    }).catch(() => {});
   180→  };
   181→
   182→  // Track previous positions so we can compute heading + speed from delta.
   183→  // We keep the last 3 fixes so we can smooth the speed value (rolling average)
   184→  // — this prevents the displayed value from jumping erratically between fixes.
   185→  const lastPosRef = useRef<{ lat: number; lng: number; t: number } | null>(null);
   186→  const recentSpeedsRef = useRef<number[]>([]);
   187→  const lastUpdateTsRef = useRef<number>(0);
   188→
   189→  const handlePosition = (pos: GeolocationPosition) => {
   190→    // Throttle: ignore fixes that arrive less than 500ms after the previous one
   191→    // (devices sometimes fire multiple fixes in rapid succession with stale data)
   192→    if (pos.timestamp - lastUpdateTsRef.current < 500 && lastUpdateTsRef.current > 0) {
   193→      return;
   194→    }
   195→    lastUpdateTsRef.current = pos.timestamp;
   196→
   197→    const prev = lastPosRef.current;
   198→    let speed: number;
   199→    let heading: number | null = pos.coords.heading;
   200→
   201→    if (prev) {
   202→      const dt = (pos.timestamp - prev.t) / 1000; // seconds
   203→      // Haversine distance between prev and current
   204→      const R = 6371000;
   205→      const toRad = (d: number) => (d * Math.PI) / 180;
   206→      const dLat = toRad(pos.coords.latitude - prev.lat);
   207→      const dLng = toRad(pos.coords.longitude - prev.lng);
   208→      const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(prev.lat)) * Math.cos(toRad(pos.coords.latitude)) * Math.sin(dLng / 2) ** 2;
   209→      const dist = 2 * R * Math.asin(Math.sqrt(h));
   210→
   211→      // Always compute speed from the position delta — this is MORE accurate
   212→      // than pos.coords.speed, which is often null on desktops/laptops and
   213→      // can be inaccurate on mobile. We cross-check with pos.coords.speed
   214→      // (when reported) and prefer the higher of the two when both indicate
   215→      // movement (this filters out false "stationary" readings).
   216→      let computedSpeed = 0;
   217→      if (dt > 0) {
   218→        // m/s → km/h, clamp to a realistic 0-80 km/h bus range
   219→        computedSpeed = Math.min(80, Math.max(0, (dist / dt) * 3.6));
   220→      }
   221→      const deviceSpeed = pos.coords.speed != null && !Number.isNaN(pos.coords.speed)
   222→        ? Math.max(0, pos.coords.speed * 3.6)
   223→        : 0;
   224→
   225→      if (dist < 2.5) {
   226→        // GPS jitter — bus is effectively stationary. Don't let small position
   227→        // wobble (±2.5m) register as speed.
   228→        speed = 0;
   229→      } else if (deviceSpeed > 1 && computedSpeed > 0.5) {
   230→        // Both readings indicate movement — trust the higher one (devices
   231→        // often under-report speed due to smoothing on their end)
   232→        speed = Math.max(deviceSpeed, computedSpeed);
   233→      } else if (computedSpeed > 0.5) {
   234→        // Device didn't report speed but we have delta-based movement
   235→        speed = computedSpeed;
   236→      } else if (deviceSpeed > 1) {
   237→        // Only device reported movement (delta too small to compute reliably)
   238→        speed = deviceSpeed;
   239→      } else {
   240→        speed = 0;
   241→      }
   242→
   243→      // Computed heading from prev → current (only if bus has moved enough
   244→      // to make the heading meaningful — otherwise keep previous heading)
   245→      if ((heading == null || Number.isNaN(heading)) && dist > 5) {
   246→        const hRad = Math.atan2(pos.coords.longitude - prev.lng, pos.coords.latitude - prev.lat);
   247→        heading = hRad * 180 / Math.PI;
   248→        if (heading < 0) heading += 360;
   249→      } else if (heading == null || Number.isNaN(heading)) {
   250→        heading = null;
   251→      }
   252→    } else {
   253→      // First fix — no previous position, fall back to device speed
   254→      speed = pos.coords.speed != null && !Number.isNaN(pos.coords.speed)
   255→        ? Math.max(0, pos.coords.speed * 3.6)
   256→        : 0;
   257→      if (heading == null || Number.isNaN(heading)) heading = null;
   258→    }
   259→
   260→    // Smooth the speed value with a rolling average of the last 3 fixes.
   261→    // This prevents the displayed number from jumping 18 → 35 → 22 between
   262→    // consecutive fixes (which looks unrealistic and confusing).
   263→    const smoothedSpeeds = [...recentSpeedsRef.current, speed].slice(-3);
   264→    recentSpeedsRef.current = smoothedSpeeds;
   265→    const smoothedSpeed = smoothedSpeeds.reduce((a, b) => a + b, 0) / smoothedSpeeds.length;
   266→
   267→    lastPosRef.current = { lat: pos.coords.latitude, lng: pos.coords.longitude, t: pos.timestamp };
   268→
   269→    const data: DriverGPSData = {
   270→      latitude: pos.coords.latitude,
   271→      longitude: pos.coords.longitude,
   272→      accuracy: pos.coords.accuracy,
   273→      speed: smoothedSpeed,
   274→      heading,
   275→      altitude: pos.coords.altitude,
   276→      timestamp: pos.timestamp,
   277→    };
   278→    setGps(data);
   279→    setError(null);
   280→    setUsingSimulated(false);
   281→    postLocation(data, false);
   282→  };
   283→
   284→  const handleError = (err: GeolocationPositionError) => {
   285→    let msg = "";
   286→    if (err.code === 1) msg = "Location permission denied. Using simulated GPS data.";
   287→    else if (err.code === 2) msg = "GPS signal unavailable. Using simulated GPS data.";
   288→    else if (err.code === 3) msg = "GPS request timed out. Using simulated GPS data.";
   289→    else msg = err.message;
   290→    setError(msg);
   291→    setUsingSimulated(true);
   292→    startSimulatedGPS();
   293→  };
   294→
   295→  const startSimulatedGPS = () => {
   296→    if (!vehicle) return;
   297→    // Use getRouteStopsWithCoords so each stop has interpolated coordinates
   298→    // between the route's main destination and RIT Campus. Without this, all
   299→    // stops fall back to the route's main coords and the bus appears stuck.
   300→    const stops = getRouteStopsWithCoords(vehicle.routeNo);
   301→    if (stops.length < 2) return;
   302→    const coordsList: Coord[] = stops.map((s) => s.coords || vehicle.coords);
   303→    coordsList[coordsList.length - 1] = RIT_CAMPUS_COORDS;
   304→    segIdxRef.current = 0;
   305→    progRef.current = 0;
   306→
   307→    const startPos = coordsList[0];
   308→    const initialData: DriverGPSData = {
   309→      latitude: startPos.lat,
   310→      longitude: startPos.lng,
   311→      accuracy: 3, // High accuracy simulated GPS — ±3m (was 8)
   312→      speed: 0,
   313→      heading: null,
   314→      altitude: 30,
   315→      timestamp: Date.now(),
   316→    };
   317→    setGps(initialData);
   318→    postLocation(initialData, true);
   319→
   320→    setTimeout(() => {
   321→      // Track previous simulated position so we can compute the actual speed
   322→      // from the distance moved per tick (rather than just a sine wave).
   323→      // Initialize with the starting position so the first delta isn't huge.
   324→      let prevSimPos: { lat: number; lng: number; t: number } = { lat: startPos.lat, lng: startPos.lng, t: Date.now() };
   325→      let prevTickTime = Date.now();
   326→      // Target speed for the simulated bus (km/h). We compute the progress
   327→      // per tick dynamically based on the segment length so the bus always
   328→      // moves at ~25 km/h regardless of whether the segment is 1km or 10km.
   329→      const TARGET_SPEED_KMH = 25;
   330→      // Update every 1 second (was 2s) — smoother movement, more "real-time" feel
   331→      simIntervalRef.current = setInterval(() => {
   332→        const segIdx = segIdxRef.current;
   333→        const a = coordsList[segIdxRef.current];
   334→        const b = coordsList[Math.min(segIdxRef.current + 1, coordsList.length - 1)];
   335→
   336→        // Compute the segment length in meters
   337→        const R = 6371000;
   338→        const toRad = (d: number) => (d * Math.PI) / 180;
   339→        const segDLat = toRad(b.lat - a.lat);
   340→        const segDLng = toRad(b.lng - a.lng);
   341→        const segH = Math.sin(segDLat / 2) ** 2 + Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(segDLng / 2) ** 2;
   342→        const segLen = 2 * R * Math.asin(Math.sqrt(segH)); // meters
   343→
   344→        // Compute progress per tick so the bus moves at ~25 km/h:
   345→        // distance_per_tick = TARGET_SPEED_KMH * 1000 / 3600 * dt = ~6.94 m/s
   346→        // progress_per_tick = distance_per_tick / segLen
   347→        const now = Date.now();
   348→        const dt = Math.max(0.1, (now - prevTickTime) / 1000); // seconds, min 0.1s
   349→        prevTickTime = now;
   350→        const distancePerTick = (TARGET_SPEED_KMH * 1000 / 3600) * dt; // meters
   351→        const progressPerTick = segLen > 0 ? distancePerTick / segLen : 0.015;
   352→
   353→        // Add gentle variation in speed (±5 km/h) by adjusting progress ±20%
   354→        const variation = 1 + Math.sin(now / 12000) * 0.2;
   355→        const prog = progRef.current + progressPerTick * variation;
   356→        if (prog >= 1) {
   357→          progRef.current = 0;
   358→          segIdxRef.current = Math.min(segIdx + 1, coordsList.length - 2);
   359→        } else {
   360→          progRef.current = prog;
   361→        }
   362→        const lat = a.lat + (b.lat - a.lat) * progRef.current;
   363→        const lng = a.lng + (b.lng - a.lng) * progRef.current;
   364→
   365→        // Compute actual speed from distance moved since the last tick.
   366→        // This guarantees the displayed speed matches the visible movement
   367→        // of the bus icon on the map.
   368→        const dLat = toRad(lat - prevSimPos.lat);
   369→        const dLng = toRad(lng - prevSimPos.lng);
   370→        const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(prevSimPos.lat)) * Math.cos(toRad(lat)) * Math.sin(dLng / 2) ** 2;
   371→        const dist = 2 * R * Math.asin(Math.sqrt(h));
   372→        // m/s → km/h
   373→        const baseSpeed = (dist / dt) * 3.6;
   374→        // Clamp to a realistic city bus speed range: 0-45 km/h
   375→        const speed = Math.max(0, Math.min(45, baseSpeed));
   376→        prevSimPos = { lat, lng, t: now };
   377→
   378→        const heading = Math.atan2(b.lng - lng, b.lat - lat) * 180 / Math.PI;
   379→
   380→        const data: DriverGPSData = {
   381→          latitude: lat,
   382→          longitude: lng,
   383→          // Higher accuracy: ±3-5m (was ±8-12m) — looks like real high-accuracy GPS
   384→          accuracy: 3 + Math.random() * 2,
   385→          speed,
   386→          heading: heading < 0 ? heading + 360 : heading,
   387→          altitude: 30 + Math.random() * 10,
   388→          timestamp: now,
   389→        };
   390→        setGps(data);
   391→        postLocation(data, true);
   392→      }, 1000);
   393→    }, 3000); // 3 second delay before bus starts moving (was 4s)
   394→  };
   395→
   396→  const startSharing = () => {
   397→    if (!vehicle) { show("Select your route first", "error"); return; }
   398→    if (!dbVehicleId) { show("Vehicle not found in database. Try again.", "error"); return; }
   399→    setError(null);
   400→    setSharing(true);
   401→    setTripStartTime(Date.now());
   402→    setGps(null);
   403→
   404→    fetch("/api/tracking/start", {
   405→      method: "POST",
   406→      headers: { "Content-Type": "application/json" },
   407→      body: JSON.stringify({ vehicleId: dbVehicleId }),
   408→    }).catch(() => {});
   409→
   410→    if (navigator.geolocation) {
   411→      // Always enable high accuracy (GPS satellite) regardless of source —
   412→      // network source just means we tolerate lower accuracy, but we still
   413→      // request the best possible fix. maximumAge: 0 means always get fresh
   414→      // position; timeout: 5000 means fail fast if GPS is slow.
   415→      watchIdRef.current = navigator.geolocation.watchPosition(handlePosition, handleError, {
   416→        enableHighAccuracy: true,
   417→        maximumAge: 0,
   418→        timeout: 5000,
   419→      });
   420→    } else {
   421→      setError("Geolocation not supported. Using simulated GPS data.");
   422→      setUsingSimulated(true);
   423→      startSimulatedGPS();
   424→    }
   425→    show("Live location sharing started");
   426→  };
   427→
   428→  const stopSharing = () => {
   429→    if (watchIdRef.current != null) {
   430→      navigator.geolocation.clearWatch(watchIdRef.current);
   431→      watchIdRef.current = null;
   432→    }
   433→    if (simIntervalRef.current) {
   434→      clearInterval(simIntervalRef.current);
   435→      simIntervalRef.current = null;
   436→    }
   437→    // Clear the last position so the next session computes speed/heading fresh
   438→    lastPosRef.current = null;
   439→    recentSpeedsRef.current = [];
   440→    lastUpdateTsRef.current = 0;
   441→    setSharing(false);
   442→    setTripStartTime(null);
   443→    setGps(null);
   444→    setUsingSimulated(false);
   445→    setError(null);
   446→    if (dbVehicleId) {
   447→      fetch("/api/tracking/stop", {
   448→        method: "POST",
   449→        headers: { "Content-Type": "application/json" },
   450→        body: JSON.stringify({ vehicleId: dbVehicleId }),
   451→      }).catch(() => {});
   452→    }
   453→    show("Location sharing stopped", "info");
   454→  };
   455→

  1225→          const a = coordsList[newSeg];

  1260→            >
  1261→              <i className="fas fa-chevron-left" />
  1262→            </button>
  1263→            <div className={`rounded-xl bg-gradient-to-br from-amber-500 to-amber-600 flex items-center justify-center flex-shrink-0 shadow-lg shadow-amber-500/20 ${isMobile ? "w-8 h-8" : "w-10 h-10"}`}>
