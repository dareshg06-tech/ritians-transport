   149→// ── Time helpers ──
   150→export function parseTime(t: string): number {
   151→  if (!t) return 0;
   152→  const [tp, mp] = t.trim().split(/\s+/);
   153→  const m = (mp || "").toLowerCase();
   154→  const [h, mn] = tp.split(".");
   155→  let hr = parseInt(h);
   156→  const mi = parseInt(mn || "0");
   157→  if (m === "pm" && hr !== 12) hr += 12;
   158→  if (m === "am" && hr === 12) hr = 0;
   159→  return hr * 60 + mi;
   160→}

   168→export function isToday(ts: number): boolean {
   169→  if (!ts) return false;
   170→  const d = new Date(ts);

   172→  return d.getDate() === n.getDate() && d.getMonth() === n.getMonth() && d.getFullYear() === n.getFullYear();
   173→}

   178→// ── Afternoon departure time from college ──
   179→export const AFTERNOON_DEPARTURE = "3.40 pm"; // 15:40 — buses leave RIT Campus
   180→export const AFTERNOON_DEPARTURE_MIN = 15 * 60 + 40; // 940 minutes from midnight
   181→export const MORNING_ARRIVAL_MIN = 7 * 60 + 40; // 460 — buses arrive RIT at 7.40 am
   182→
   183→/**
   184→ * Convert minutes-from-midnight → "H.MM am/pm" string.
   185→ */
   186→export function formatTime(min: number): string {
   187→  if (min < 0) min += 24 * 60;
   188→  if (min >= 24 * 60) min -= 24 * 60;
   189→  let hr = Math.floor(min / 60);
   190→  const mi = min % 60;
   191→  const suffix = hr >= 12 ? "pm" : "am";
   192→  let h12 = hr % 12;
   193→  if (h12 === 0) h12 = 12;
   194→  return `${h12}.${mi.toString().padStart(2, "0")} ${suffix}`;
   195→}

   197→/**
   198→ * Calculate the afternoon return-trip schedule for a route.
   199→ *
   200→ * Morning: bus starts at stop[0] at `route.start`, arrives RIT Campus at 7.40 am.
   201→ * Afternoon: bus departs RIT Campus at 3.40 pm, then visits each prior stop in REVERSE.
   202→ *
   203→ * For each reversed stop, the afternoon arrival time =
   204→ *   3.40 pm + (7.40 am − morning_time_at_that_stop)
   205→ *
   206→ * Example: R01 Lift Gate morning = 5.50 am → afternoon arrival =
   207→ *   3.40 pm + (7.40 am − 5.50 am) = 3.40 pm + 1h50m = 5.30 pm
   208→ */
   209→export function getReturnTrip(routeNo: string): Stop[] {
   210→  const morning = routeStops[routeNo];
   211→  if (!morning || morning.length === 0) return [];
   212→  // Reverse the list (so RIT Campus becomes first), then compute afternoon times.
   213→  // Note: morning list already ends at "RIT Campus" at 7.40 am.
   214→  const reversed = [...morning].reverse();
   215→  return reversed.map((s) => {
   216→    const mMin = parseTime(s.time);
   217→    const offset = MORNING_ARRIVAL_MIN - mMin; // how long from this stop → RIT (morning)
   218→    const afternoon = AFTERNOON_DEPARTURE_MIN + offset;
   219→    return { stop: s.stop, time: formatTime(afternoon), coords: s.coords };
   220→  });
   221→}
   222→
   223→/**
   224→ * Total afternoon trip duration (minutes) for a route.
   225→ */
   226→export function getReturnTripDuration(routeNo: string): number {
   227→  const morning = routeStops[routeNo];
   228→  if (!morning || morning.length === 0) return 0;
   229→  const firstStop = morning[0];
   230→  return MORNING_ARRIVAL_MIN - parseTime(firstStop.time);
   231→}
   232→
   233→/**
   234→ * Afternoon arrival time at the last stop (the route's main destination).
   235→ */
   236→export function getReturnArrival(routeNo: string): string {
   237→  const rt = getReturnTrip(routeNo);
   238→  if (!rt.length) return "—";
   239→  return rt[rt.length - 1].time;
   240→}
