function parseTime(t){
  if(!t)return 0;
  const [tp,mp]=(t.trim().split(/\s+/));
  const m=(mp||'').toLowerCase();
  const,mn]=tp.split('.');
  let hr=parseInt(h);const mi=parseInt(mn||0);
  if(m==='pm'&&hr!==12)hr+=12;
  if(m==='am'&&hr===12)hr=0;
  return hr*60+mi;
}

function toast(msg,type='success'){
  if(window.RitiansToast && typeof window.RitiansToast.show==='function'){
    window.RitiansToast.show(msg,type);
  }else{
    console.error('RitiansToast not loaded — check that toast.js is present and loaded before this script.');
    // Minimal fallback so the user still sees feedback. Don't rely on
    // #toastContainer already existing — it's normally created by
    // toast.js's own getContainer(), which never ran if we're in this
    // branch, so build it here instead of silently no-oping.
    let c=document.getElementById('toastContainer');
    if(!c){
      c=document.createElement('div');
      c.id='toastContainer';
      c.style.cssText='position:fixed;bottom:24px;left:50%;transform:translateX(-50%);z-index:9999;display:flex;flex-direction:column;gap:8px;';
      document.body.appendChild(c);
    }
    const el=document.createElement('div');
    el.className='toast-fallback toast-'+type;
    el.textContent=typeof msg==='string'?msg:(msg&&msg.title)||(msg&&msg.message)||'Notification';
    el.style.cssText='background:#1a1f2e;color:#fff;padding:10px 16px;border-radius:8px;font-size:14px;box-shadow:0 4px 20px rgba(0,0,0,0.4);border:1px solid rgba(255,255,255,0.1);';
    c.appendChild(el);
    setTimeout(()=>el.remove(),3000);
  }
}

function renderWeekStrip(){
  const c=document.getElementById('weekDays');
  const today=new Date();today.setHours(0,0,0,0);
  let html='';
  for(let i=0;i<7;i++){
    const d=new Date(today);d.setDate(today.getDate()+i);
    const lbl=d.toLocaleDateString('en-IN',{weekday:'short'}).toUpperCase()+' '+d.getDate();
    const active=selectedViewDate.toDateString()===d.toDateString();
    html+=`<button class="wday${active?' active':''}" data-iso="${d.toISOString()}">${lbl}</button>`;
  }
  c.innerHTML=html;
  c.querySelectorAll('.wday').forEach(b=>b.addEventListener('click',()=>{
    selectedViewDate=new Date(b.dataset.iso);
    renderWeekStrip();
    updateDateLabel();
  }));
}
