  1090→function parseTime(t){
  1091→  if(!t)return 0;
  1092→  const [tp,mp]=(t.trim().split(/\s+/));
  1093→  const m=(mp||'').toLowerCase();
  1094→  const[h,mn]=tp.split('.');
  1095→  let hr=parseInt(h);const mi=parseInt(mn||0);
  1096→  if(m==='pm'&&hr!==12)hr+=12;
  1097→  if(m==='am'&&hr===12)hr=0;
  1098→  return hr*60+mi;
  1099→}

  1631→// Two delivery paths call into this same logic:
  1632→//   1. Capacitor's PushNotifications 'pushNotificationReceived' listener

  1634→//   2. window.__rtHandleNativePush(data), called directly by
  1635→//      RtMessagingService.java via webView.evaluateJavascript(). This is
  1636→//      now the PRIMARY path: Android does not reliably deliver one FCM
  1637→//      message to multiple registered FirebaseMessagingService classes, and
  1638→//      RtMessagingService (which builds the tray notification) is the one
  1639→//      that actually receives messages in this app — Capacitor's own
  1640→//      MessagingService was being starved out, so its JS listener alone

  1657→  const title = data.title || '';

  1660→
  1661→  if (!showBusStartedToast(data)) {
