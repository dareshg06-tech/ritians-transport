   301→    if (stops.length < 2) return;
   302→    const coordsList: Coord[] = stops.map((s) => s.coords || vehicle.coords);
   303→    coordsList[coordsList.length - 1] = RIT_CAMPUS_COORDS;
   304→    segIdxRef.current = 0;
   305→    progRef.current = 0;
   306→
   307→    const startPos = coordsList[0];
   308→    const initialData: DriverGPSData = {
   309→      latitude: startPos.lat,
   310→      longitude: startPos.lng,
   311→      accuracy: 3, // High accuracy simulated GPS — ±3m (was 8)
   312→      speed: 0,
   313→      heading: null,
   314→      altitude: 30,
   315→      timestamp: Date.now(),
   316→    };
   317→    setGps(initialData);
   318→    postLocation(initialData, true);
   319→
   320→    setTimeout(() => {
   321→      // Track previous simulated position so we can compute the actual speed
   322→      // from the distance moved per tick (rather than just a sine wave).
   323→      // Initialize with the starting position so the first delta isn't huge.
   324→      let prevSimPos: { lat: number; lng: number; t: number } = { lat: startPos.lat, lng: startPos.lng, t: Date.now() };
   325→      let prevTickTime = Date.now();
   326→      // Target speed for the simulated bus (km/h). We compute the progress
   327→      // per tick dynamically based on the segment length so the bus always
   328→      // moves at ~25 km/h regardless of whether the segment is 1km or 10km.
   329→      const TARGET_SPEED_KMH = 25;
   330→      // Update every 1 second (was 2s) — smoother movement, more "real-time" feel
   331→      simIntervalRef.current = setInterval(() => {
   332→        const segIdx = segIdxRef.current;
   333→        const a = coordsList[segIdxRef.current];
   334→        const b = coordsList[Math.min(segIdxRef.current + 1, coordsList.length - 1)];
   335→
   336→        // Compute the segment length in meters
   337→        const R = 6371000;
   338→        const toRad = (d: number) => (d * Math.PI) / 180;
   339→        const segDLat = toRad(b.lat - a.lat);
   340→        const segDLng = toRad(b.lng - a.lng);
   341→        const segH = Math.sin(segDLat / 2) ** 2 + Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(segDLng / 2) ** 2;
   342→        const segLen = 2 * R * Math.asin(Math.sqrt(segH)); // meters
   343→
   344→        // Compute progress per tick so the bus moves at ~25 km/h:
   345→        // distance_per_tick = TARGET_SPEED_KMH * 1000 / 3600 * dt = ~6.94 m/s
   346→        // progress_per_tick = distance_per_tick / segLen
   347→        const now = Date.now();
   348→        const dt = Math.max(0.1, (now - prevTickTime) / 1000); // seconds, min 0.1s
   349→        prevTickTime = now;
   350→        const distancePerTick = (TARGET_SPEED_KMH * 1000 / 3600) * dt; // meters
   351→        const progressPerTick = segLen > 0 ? distancePerTick / segLen : 0.015;
   352→
   353→        // Add gentle variation in speed (±5 km/h) by adjusting progress ±20%
   354→        const variation = 1 + Math.sin(now / 12000) * 0.2;
   355→        const prog = progRef.current + progressPerTick * variation;
   356→        if (prog >= 1) {
   357→          progRef.current = 0;
   358→          segIdxRef.current = Math.min(segIdx + 1, coordsList.length - 2);
   359→        } else {
   360→          progRef.current = prog;
   361→        }
   362→        const lat = a.lat + (b.lat - a.lat) * progRef.current;
   363→        const lng = a.lng + (b.lng - a.lng) * progRef.current;
   364→
   365→        // Compute actual speed from distance moved since the last tick.
   366→        // This guarantees the displayed speed matches the visible movement
   367→        // of the bus icon on the map.
   368→        const dLat = toRad(lat - prevSimPos.lat);
   369→        const dLng = toRad(lng - prevSimPos.lng);
   370→        const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(prevSimPos.lat)) * Math.cos(toRad(lat)) * Math.sin(dLng / 2) ** 2;
   371→        const dist = 2 * R * Math.asin(Math.sqrt(h));
   372→        // m/s → km/h
   373→        const baseSpeed = (dist / dt) * 3.6;
   374→        // Clamp to a realistic city bus speed range: 0-45 km/h
   375→        const speed = Math.max(0, Math.min(45, baseSpeed));
   376→        prevSimPos = { lat, lng, t: now };
   377→
   378→        const heading = Math.atan2(b.lng - lng, b.lat - lat) * 180 / Math.PI;
   379→
   380→        const data: DriverGPSData = {
   381→          latitude: lat,
   382→          longitude: lng,
   383→          // Higher accuracy: ±3-5m (was ±8-12m) — looks like real high-accuracy GPS
   384→          accuracy: 3 + Math.random() * 2,
   385→          speed,
   386→          heading: heading < 0 ? heading + 360 : heading,
   387→          altitude: 30 + Math.random() * 10,
   388→          timestamp: now,
   389→        };
   390→        setGps(data);
   391→        postLocation(data, true);
   392→      }, 1000);
   393→    }, 3000); // 3 second delay before bus starts moving (was 4s)
   394→  };
   395→
   396→  const startSharing = () => {
   397→    if (!vehicle) { show("Select your route first", "error"); return; }
   398→    if (!dbVehicleId) { show("Vehicle not found in database. Try again.", "error"); return; }
   399→    setError(null);
   400→    setSharing(true);
   401→    setTripStartTime(Date.now());
   402→    setGps(null);
   403→
   404→    fetch("/api/tracking/start", {
   405→      method: "POST",
   406→      headers: { "Content-Type": "application/json" },
   407→      body: JSON.stringify({ vehicleId: dbVehicleId }),
   408→    }).catch(() => {});
   409→
   410→    if (navigator.geolocation) {
   411→      // Always enable high accuracy (GPS satellite) regardless of source —
   412→      // network source just means we tolerate lower accuracy, but we still
   413→      // request the best possible fix. maximumAge: 0 means always get fresh
   414→      // position; timeout: 5000 means fail fast if GPS is slow.
   415→      watchIdRef.current = navigator.geolocation.watchPosition(handlePosition, handleError, {
   416→        enableHighAccuracy: true,
   417→        maximumAge: 0,
   418→        timeout: 5000,
   419→      });
   420→    } else {
   421→      setError("Geolocation not supported. Using simulated GPS data.");

   423→      startSimulatedGPS();
   424→    }
   425→    show("Live location sharing started");
   426→  };
   427→

   430→      navigator.geolocation.clearWatch(watchIdRef.current);
   431→      watchIdRef.current = null;
   432→    }
   433→    if (simIntervalRef.current) {
   434→      clearInterval(simIntervalRef.current);
   435→      simIntervalRef.current = null;
   436→    }
   437→    // Clear the last position so the next session computes speed/heading fresh
   438→    lastPosRef.current = null;
   439→    recentSpeedsRef.current = [];
   440→    lastUpdateTsRef.current = 0;
   441→    setSharing(false);
   442→    setTripStartTime(null);
   443→    setGps(null);
   444→    setUsingSimulated(false);
   445→    setError(null);
   446→    if (dbVehicleId) {
   447→      fetch("/api/tracking/stop", {
   448→        method: "POST",
   449→        headers: { "Content-Type": "application/json" },
   450→        body: JSON.stringify({ vehicleId: dbVehicleId }),
   451→      }).catch(() => {});
   452→    }
   453→    show("Location sharing stopped", "info");
   454→  };
   455→
   456→  useEffect(() => {
   457→    return () => {
   458→      if (watchIdRef.current != null) navigator.geolocation.clearWatch(watchIdRef.current);
   459→      if (simIntervalRef.current) clearInterval(simIntervalRef.current);
   460→    };
   461→  }, []);
   462→
   463→  const tripDuration = tripStartTime ? Math.floor((Date.now() - tripStartTime) / 60000) : 0;
   464→
   465→  return (
   466→    <div className="space-y-3">
   467→      <div className="flex items-center justify-between gap-3 pt-1 pb-2">
   468→        <div className="min-w-0">
   469→          <div className="text-[10px] uppercase tracking-widest text-amber-400 font-bold">Driver Mode</div>
   470→          <h2 className="text-lg font-bold text-white leading-tight mt-0.5 truncate">Broadcast your GPS</h2>
   471→          <div className="text-[10px] text-slate-500">Pushes live location every 2 seconds to passengers</div>
   472→        </div>
   473→        <div className="flex items-center gap-2 flex-shrink-0">
   474→          <span className={`inline-flex items-center justify-center rounded-md border px-2 py-0.5 text-xs font-medium gap-1.5 ${
   475→            sharing
   476→              ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-300"
   477→              : "border-slate-600 bg-slate-700/30 text-slate-400"
   478→          }`}>
   479→            <i className={`fas fa-circle text-[6px] ${sharing ? "animate-pulse" : ""}`} />
   480→            {sharing ? "BROADCASTING" : "IDLE"}
   481→          </span>
   482→          <div className="w-9 h-9 rounded-full bg-gradient-to-br from-amber-500 to-amber-600 flex items-center justify-center text-white text-sm font-bold">
   483→            D
   484→          </div>
   485→        </div>
   486→      </div>
   487→
   488→      {/* Route info banner */}
   489→      {vehicle && (

   492→          <span className="text-amber-200 flex-1 min-w-0 truncate">
   493→            <strong>BUS {String(vehicle.no).padStart(2, "0")}</strong> · {vehicle.routeName} → RIT Campus

   499→      {/* Info alert */}
   500→      <div className="rounded-xl px-3.5 py-3 border border-[#1f2538] bg-[#10131f] text-[11px] text-slate-400 flex items-start gap-2">

   541→          >
   542→            <i className="fas fa-satellite block text-base mb-1 text-cyan-400" />

   550→                ? "border-cyan-500/40 bg-cyan-500/10"

   574→        >
   575→          <i className="fas fa-stop" /> Stop Sharing Location

   578→
   579→      {/* Error message */}
   580→      {error && !usingSimulated && (

   685→                <i className="fas fa-map" /> {sharing ? "Live Bus Position" : "Route Preview"}
   686→              </div>

   694→                    LIVE · {tick}s ago
